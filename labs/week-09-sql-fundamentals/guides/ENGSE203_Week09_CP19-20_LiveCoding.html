เปิดการตรวจ Foreign Key ก่อนเสมอ

VS Code + SQLite

ทำไมย้ายจาก playground มา VS Code — playground บางเจ้า syntax ต่างจาก SQLite แท้ ไม่ควรติดนิสัย · และ VS Code คือเครื่องมือที่ใช้จริงสัปดาห์หน้า

PRAGMA foreign_keys = ON;
⚠ ข้อนี้สำคัญที่สุดของช่วงนี้ — SQLite ปิดการตรวจ FK ไว้เป็นค่าเริ่มต้น · ถ้าไม่เปิด คำสั่งที่ควรถูกปฏิเสธจะผ่านหมด · ต้องสั่งทุกครั้งที่เปิดฐานข้อมูล

สร้างตาราง users

schema.sql

ผู้แจ้งคำร้อง — 4 คอลัมน์

CREATE TABLE users (
  id          INTEGER PRIMARY KEY AUTOINCREMENT,
  name        TEXT NOT NULL,
  department  TEXT NOT NULL,
  email       TEXT NOT NULL UNIQUE
);
③

สร้างตาราง requests

schema.sql

คำร้อง — สังเกต CHECK, DEFAULT และ FOREIGN KEY

CREATE TABLE requests (
  id            TEXT PRIMARY KEY,
  requester_id  INTEGER NOT NULL,
  request_type  TEXT NOT NULL
                CHECK (request_type IN ('แจ้งซ่อม','บริการบัญชีผู้ใช้','ขอใช้อุปกรณ์','อื่น ๆ')),
  location      TEXT NOT NULL,
  details       TEXT NOT NULL,
  priority      TEXT NOT NULL DEFAULT 'normal'
                CHECK (priority IN ('normal','urgent')),
  status        TEXT NOT NULL DEFAULT 'pending'
                CHECK (status IN ('pending','in-progress','completed')),
  created_at    TEXT NOT NULL DEFAULT (datetime('now','localtime')),
  FOREIGN KEY (requester_id) REFERENCES users(id)
);
3 อย่างที่ต้องสังเกต — CHECK จำกัดค่า · DEFAULT เติมให้เมื่อไม่ใส่ · FOREIGN KEY บังคับว่า requester_id ต้องมีจริง

⑤

CRUD — เพิ่ม แก้ ลบ

สามคำสั่งตรงกับ POST / PUT / DELETE ที่ทำใน API

-- INSERT (ไม่ใส่ priority/status → ใช้ DEFAULT)
INSERT INTO requests (id, requester_id, request_type, location, details)
VALUES ('REQ-006', 2, 'แจ้งซ่อม', 'ห้อง 401', 'ไฟกะพริบ');

-- UPDATE (⚠ ต้องมี WHERE!)
UPDATE requests SET status = 'in-progress' WHERE id = 'REQ-006';

-- DELETE
DELETE FROM requests WHERE id = 'REQ-006';
⚠ กับดักอันตรายที่สุด — ลืม WHERE
UPDATE requests SET status = 'completed'; จะเปลี่ยนทุกแถว ไม่มี error และย้อนกลับไม่ได้
นิสัย — SELECT ด้วยเงื่อนไขเดียวกันก่อนเสมอ
⑥

ลองทำผิด — ให้เห็น constraint ทำงาน

ทั้ง 3 คำสั่งนี้ต้องถูกปฏิเสธ

-- FK: ไม่มีผู้ใช้ id=999
INSERT INTO requests (id,requester_id,request_type,location,details)
VALUES ('X',999,'แจ้งซ่อม','a','ทดสอบระบบ');

-- CHECK: status นอกรายการ
UPDATE requests SET status = 'ยกเลิก' WHERE id = 'REQ-001';

-- FK: ลบผู้ใช้ที่ยังมีคำร้อง
DELETE FROM users WHERE id = 1;
ถ้าข้อแรกผ่านได้ — แปลว่าลืม PRAGMA foreign_keys = ON
