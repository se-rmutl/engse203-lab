# สร้างไฟล์ทดลอง try.mjs แล้วรัน
node --disable-warning=ExperimentalWarning try.mjs
// try.mjs — สนามซ้อม SQL แบบออฟไลน์
import { DatabaseSync } from 'node:sqlite';
const db = new DatabaseSync(':memory:');   // เก็บใน RAM ไม่สร้างไฟล์

db.exec(`
  CREATE TABLE requests (id TEXT, status TEXT, priority TEXT);
  INSERT INTO requests VALUES ('REQ-001','pending','urgent');
  INSERT INTO requests VALUES ('REQ-002','completed','normal');
`);

// เปลี่ยน SQL บรรทัดนี้แล้วรันใหม่ได้เรื่อย ๆ
console.table(db.prepare("SELECT * FROM requests WHERE status = 'pending'").all());

1.1 ปัญหา 4 อย่างของไฟล์ JSON

ปัญหาเกิดอะไรขึ้น
ช้าเมื่อข้อมูลเยอะต้องอ่านทั้งไฟล์ทุกครั้ง แม้ต้องการแค่รายการเดียว
เขียนพร้อมกันแล้วพังถ้าสองคนบันทึกพร้อมกัน คนหลังจะเขียนทับคนแรก — ข้อมูลหาย
ไม่มีใครตรวจความถูกต้องใส่ "status": "ยกเลิก" ก็ได้ ทั้งที่ระบบไม่รู้จักสถานะนี้
ข้อมูลซ้ำและไม่ตรงกันชื่อผู้แจ้งเก็บซ้ำในทุกคำร้อง · แก้ไม่ครบแล้วจะมีหลายเวอร์ชัน
🟪 เทียบกับของที่คุ้น

ไฟล์ JSON = สมุดจดเล่มเดียว — จดได้เร็ว แต่พอจดเยอะ ๆ จะหาอะไรทีต้องพลิกทั้งเล่ม และถ้าสองคนจะเขียนพร้อมกันก็ต้องแย่งสมุดกัน
ฐานข้อมูล = ตู้เอกสารที่มีระบบ — มีช่องแยก มีดัชนี หลายคนหยิบพร้อมกันได้ และมีกฎว่าเอกสารแบบไหนใส่ช่องไหนได้บ้าง

1.2 ฐานข้อมูลให้อะไรเพิ่ม

ความสามารถแก้ปัญหาข้อไหน
ค้นหาเฉพาะที่ต้องการโดยไม่อ่านทั้งหมดช้าเมื่อข้อมูลเยอะ
จัดการการเขียนพร้อมกันให้เองเขียนพร้อมกันแล้วพัง
บังคับกฎของข้อมูลเอง (constraint)ไม่มีใครตรวจความถูกต้อง
เชื่อมข้อมูลข้ามตารางได้ (relationship)ข้อมูลซ้ำและไม่ตรงกัน

3.1 ชนิดข้อมูลใน SQLite

ชนิดเก็บอะไรตัวอย่างในโปรเจกต์เรา
TEXTข้อความname · location · status
INTEGERจำนวนเต็มusers.id · requester_id
REALทศนิยมยังไม่ได้ใช้
BLOBข้อมูลดิบ เช่น รูปภาพยังไม่ได้ใช้

3.2 ข้อกำหนด (Constraint) — กฎที่ฐานข้อมูลบังคับเอง

Constraint

กฎที่กำหนดไว้ตอนสร้างตาราง · ฐานข้อมูลจะปฏิเสธข้อมูลที่ผิดกฎเอง ไม่ว่าจะใส่เข้ามาจากทางไหน

Constraintบังคับอะไรใช้ในโปรเจกต์เรา
PRIMARY KEYไม่ซ้ำ + ไม่เป็น NULLusers.id · requests.id
NOT NULLต้องมีค่าเสมอname · location · details
UNIQUEห้ามซ้ำกับแถวอื่นusers.email
DEFAULTถ้าไม่ใส่ ใช้ค่านี้status = 'pending'
CHECKค่าต้องตรงเงื่อนไขstatus ต้องเป็น 1 ใน 3 ค่า
FOREIGN KEYต้องมีอยู่จริงในอีกตารางrequester_id → users.id

3.3 ตัวอย่างจริง

CREATE TABLE requests (
  id            TEXT PRIMARY KEY,
  requester_id  INTEGER NOT NULL,
  request_type  TEXT NOT NULL
                CHECK (request_type IN ('แจ้งซ่อม','บริการบัญชีผู้ใช้','ขอใช้อุปกรณ์','อื่น ๆ')),
  location      TEXT NOT NULL,
  details       TEXT NOT NULL,
  priority      TEXT NOT NULL DEFAULT 'normal'
                CHECK (priority IN ('normal','urgent')),
  status        TEXT NOT NULL DEFAULT 'pending'
                CHECK (status IN ('pending','in-progress','completed')),
  created_at    TEXT NOT NULL DEFAULT (datetime('now','localtime')),

  FOREIGN KEY (requester_id) REFERENCES users(id)
);
🟩 ทายก่อนอ่านต่อ

เรามี validation ฝั่ง API อยู่แล้วตั้งแต่ Week 06 CP04 — แล้วทำไมยังต้องมี CHECK ในฐานข้อมูลอีก

เฉลย

เพราะ API ไม่ใช่ทางเดียวที่ข้อมูลเข้าฐานข้อมูลได้

ข้อมูลอาจเข้ามาทาง — สคริปต์นำเข้าข้อมูล · คนแก้ด้วยมือผ่านเครื่องมือ · ระบบอื่นที่มาเชื่อมทีหลัง · หรือโค้ดที่เขียนผิดในอนาคต

การป้องกันสองชั้นดีกว่าชั้นเดียว — validation ที่ API ให้ข้อความดี ๆ กับผู้ใช้ ส่วน constraint ที่ฐานข้อมูลเป็นด่านสุดท้ายที่ไม่มีใครข้ามได้

🟥 SQLite ปิดการตรวจ Foreign Key ไว้เป็นค่าเริ่มต้น

ต้องสั่ง PRAGMA foreign_keys = ON; ทุกครั้งที่เปิดฐานข้อมูล ไม่งั้น constraint จะไม่ทำงาน และใส่ requester_id มั่วเข้าไปได้
เป็นข้อจำกัดทางประวัติศาสตร์ของ SQLite เพื่อให้เข้ากันได้กับโปรแกรมเก่า

สรุปบทที่ 3
  • SQLite มี 4 ชนิดหลัก — TEXT · INTEGER · REAL · BLOB
  • constraint คือกฎที่ฐานข้อมูลบังคับเอง ไม่ว่าข้อมูลจะเข้ามาทางไหน
  • มี validation ที่ API แล้วก็ยังควรมี constraint — ป้องกันสองชั้น
  • ต้องสั่ง PRAGMA foreign_keys = ON ไม่งั้น FK ไม่ทำงาน

SELECT — อ่านข้อมูลออกมา

เวลาอ่าน ≈ 11 นาที

SQL (Structured Query Language)

ภาษาที่ใช้สั่งงานฐานข้อมูล · บอกว่าอยากได้อะไร ไม่ต้องบอกว่าหาอย่างไร — ฐานข้อมูลคิดวิธีที่เร็วที่สุดให้เอง

🟪 ต่างจาก JavaScript อย่างไร

ใน JavaScript เราบอกขั้นตอน — วนลูป เช็คทีละตัว เก็บใส่ array ใหม่

// JavaScript — บอกวิธีทำ
const result = requests.filter(r => r.status === 'pending');
-- SQL — บอกแค่ว่าอยากได้อะไร
SELECT * FROM requests WHERE status = 'pending';

ฐานข้อมูลจะตัดสินใจเองว่าจะใช้ index ไหม จะอ่านแบบไหนถึงเร็วที่สุด — เราไม่ต้องคิดแทนมัน

4.1 รูปแบบพื้นฐาน

SELECT คอลัมน์ที่อยากได้
FROM   ตารางไหน;
-- เอาทุกคอลัมน์
SELECT * FROM requests;

-- เอาเฉพาะบางคอลัมน์
SELECT id, location, status FROM requests;
🟥 SELECT * สะดวก แต่ไม่ควรใช้ในโค้ดจริง

เพราะถ้ามีคนเพิ่มคอลัมน์ใหม่ในตาราง โค้ดของคุณจะได้ข้อมูลเกินมาโดยไม่รู้ตัว · และถ้าคอลัมน์นั้นมีข้อมูลอ่อนไหว เช่น รหัสผ่าน ก็จะหลุดออกไปด้วย
ใช้ * ตอนสำรวจข้อมูล · ระบุชื่อคอลัมน์ตอนเขียนโค้ดจริง

4.2 ตั้งชื่อใหม่ให้คอลัมน์ด้วย AS

SELECT id,
       request_type AS requestType,
       created_at   AS createdAt
FROM requests;
🟨 ทำไมเรื่องนี้สำคัญกับเรา

ในฐานข้อมูลนิยมตั้งชื่อแบบ request_type (มีขีดล่าง) · แต่ใน JavaScript นิยม requestType
AS ช่วยแปลงตอนดึงออกมา — สัปดาห์หน้าจะได้ใช้เรื่องนี้ตอนทำให้ API คืนข้อมูลรูปแบบเดิมที่ React ใช้อยู่

4.3 ค่าที่ไม่ซ้ำด้วย DISTINCT

-- มีสถานะอะไรใช้อยู่บ้าง
SELECT DISTINCT status FROM requests;

4.4 จำกัดจำนวนด้วย LIMIT

SELECT * FROM requests LIMIT 5;
🟨 เคล็ดลับตอนสำรวจข้อมูลใหม่

เจอตารางที่ไม่เคยเห็น ให้เริ่มด้วย SELECT * FROM ชื่อตาราง LIMIT 5; เสมอ — เห็นหน้าตาข้อมูลโดยไม่ต้องรอโหลดทั้งหมด

  • SELECT คอลัมน์ FROM ตาราง คือรูปแบบพื้นฐาน
  • SELECT * ใช้ตอนสำรวจ · ระบุชื่อคอลัมน์ตอนเขียนโค้ดจริง
  • AS เปลี่ยนชื่อคอลัมน์ตอนดึงออกมา — จะได้ใช้จริงสัปดาห์หน้า

WHERE และ ORDER BY — กรองและเรียง

เวลาอ่าน ≈ 12 นาที

5.1 WHERE — เลือกเฉพาะแถวที่ต้องการ

SELECT * FROM requests
WHERE status = 'pending';
ตัวดำเนินการความหมายตัวอย่าง
=เท่ากับstatus = 'pending'
<> หรือ !=ไม่เท่ากับstatus <> 'completed'
> < >= <=มากกว่า/น้อยกว่าcreated_at >= '2026-09-01'
ANDต้องจริงทั้งสองข้อpriority = 'urgent' AND status = 'pending'
ORจริงข้อใดข้อหนึ่งstatus = 'pending' OR status = 'in-progress'
INอยู่ในรายการstatus IN ('pending','in-progress')
LIKEตรงกับรูปแบบdetails LIKE '%ไม่ทำงาน%'
IS NULLไม่มีค่าemail IS NULL
🟥 กับดัก 3 อันของ WHERE

① ใช้ == แบบ JavaScript — SQL ใช้ = ตัวเดียวในการเปรียบเทียบ · เขียน == จะ error

② ใช้ = NULL — ต้องใช้ IS NULL เท่านั้น · = NULL จะไม่เจออะไรเลยและไม่มี error บอก เพราะ NULL แปลว่า "ไม่รู้ค่า" ซึ่งเทียบเท่ากับอะไรไม่ได้

③ ใช้เครื่องหมายคำพูดคู่ — SQL ใช้ เครื่องหมายคำพูดเดี่ยว 'pending' สำหรับข้อความ · คำพูดคู่ "..." ใช้กับชื่อคอลัมน์ ไม่ใช่ค่า

5.2 LIKE — ค้นจากคำบางส่วน

รูปแบบหมายถึงตรงกับ
'%ไม่ทำงาน%'มีคำนี้อยู่ตรงไหนก็ได้"เครื่องปรับอากาศไม่ทำงานตั้งแต่เช้า"
'REQ-%'ขึ้นต้นด้วยREQ-001 · REQ-002
'%301'ลงท้ายด้วยห้องปฏิบัติการ 301
🟨 % คือ "อะไรก็ได้ กี่ตัวก็ได้"

เหมือน * ที่ใช้ค้นไฟล์ในคอมพิวเตอร์ · ถ้าไม่ใส่ % เลย LIKE 'ไม่ทำงาน' จะหาเฉพาะช่องที่มีข้อความนี้เป๊ะ ๆ ทั้งช่อง ซึ่งมักไม่ใช่สิ่งที่ต้องการ

5.3 ORDER BY — เรียงลำดับ

-- เรียงจากน้อยไปมาก (ค่าเริ่มต้น)
SELECT * FROM requests ORDER BY created_at;

-- เรียงจากมากไปน้อย
SELECT * FROM requests ORDER BY created_at DESC;

-- เรียงหลายชั้น — เร่งด่วนก่อน แล้วค่อยเรียงตามรหัส
SELECT * FROM requests ORDER BY priority DESC, id;
🟩 ทายก่อนอ่านต่อ

ORDER BY priority DESC จะเรียง 'urgent' กับ 'normal' อย่างไร และทำไม

เฉลย

urgent มาก่อน normal — เพราะเรียงตามตัวอักษรจากหลังไปหน้า และ u มาหลัง n

ที่ได้ผลถูกต้องเป็นความบังเอิญ — ถ้าวันหนึ่งเพิ่มระดับ 'critical' ที่ควรมาก่อน urgent จะเรียงผิดทันที เพราะ c มาก่อน n และ u
วิธีที่ถูกคือเก็บลำดับความสำคัญเป็นตัวเลขแยกไว้ แต่เกินขอบเขตสัปดาห์นี้

5.4 ลำดับของคำสั่งใน SQL

SELECT   คอลัมน์
FROM     ตาราง
WHERE    เงื่อนไข
ORDER BY คอลัมน์ที่ใช้เรียง
LIMIT    จำนวน;
🟥 สลับลำดับไม่ได้

SQL บังคับลำดับนี้ตายตัว · เขียน ORDER BY ก่อน WHERE จะ error ทันที
จำง่าย ๆ — เอาอะไร (SELECT) · จากไหน (FROM) · กรองยังไง (WHERE) · เรียงยังไง (ORDER BY) · เอากี่อัน (LIMIT)

5.5 ลำดับการทำงานจริงของฐานข้อมูล

เราเขียน SELECT ไว้บรรทัดแรก — แต่ฐานข้อมูลทำเป็นขั้นสุดท้าย

สรุปบทที่ 5
  • WHERE กรองแถว · ใช้ = ตัวเดียว และคำพูดเดี่ยว สำหรับข้อความ
  • NULL ต้องใช้ IS NULL ไม่ใช่ = NULL
  • LIKE กับ % ค้นจากคำบางส่วน
  • ORDER BY เรียงได้หลายชั้น · DESC คือมากไปน้อย
  • ลำดับคำสั่งบังคับตายตัว — SELECT · FROM · WHERE · ORDER BY · LIMIT
  • แต่ฐานข้อมูลทำงานคนละลำดับ — FROM → WHERE → ORDER BY → SELECT

ฐานข้อมูลจะเติม 'normal', 'pending' และเวลาปัจจุบันให้เอง · นี่คือประโยชน์ของ DEFAULT ที่เรียนในบทที่ 3

6.2 UPDATE — แก้ไขแถวที่มีอยู่

UPDATE requests
SET    status = 'in-progress'
WHERE  id = 'REQ-001';
🟥 กับดักที่อันตรายที่สุดในบทนี้ — ลืม WHERE 
UPDATE requests SET status = 'completed';   -- ไม่มี WHERE

คำสั่งนี้จะ เปลี่ยนสถานะของคำร้องทุกรายการในระบบเป็น completed · ไม่มี error ไม่มีคำเตือน และย้อนกลับไม่ได้
นิสัยที่ควรสร้าง — เขียน WHERE ก่อนเสมอ แล้วค่อยเขียน SET ทีหลัง

🟩 วิธีป้องกันที่ใช้ได้จริง

ลอง SELECT ด้วยเงื่อนไขเดียวกันก่อนเสมอ

-- ① ดูก่อนว่าจะกระทบกี่แถว
SELECT * FROM requests WHERE id = 'REQ-001';

-- ② ถ้าถูกต้องแล้วค่อยเปลี่ยนเป็น UPDATE
UPDATE requests SET status = 'in-progress' WHERE id = 'REQ-001';

6.3 DELETE — ลบแถว

DELETE FROM requests WHERE id = 'REQ-006';
🟥 DELETE FROM requests; ลบทุกแถว

อันตรายกว่า UPDATE อีก เพราะข้อมูลหายไปเลย · ใช้วิธีเดียวกัน — SELECT ดูก่อนเสมอ

🟩 ทายก่อนอ่านต่อ

ถ้าลอง DELETE FROM users WHERE id = 1; ทั้งที่สมชาย (id=1) มีคำร้องอยู่ 2 รายการ — จะเกิดอะไรขึ้น

เฉลย

ฐานข้อมูลจะปฏิเสธ ด้วยข้อความ FOREIGN KEY constraint failed

เพราะถ้าลบสมชายออกไป คำร้อง REQ-001 และ REQ-004 จะชี้ไปหาคนที่ไม่มีอยู่จริง — กลายเป็นข้อมูลกำพร้า
นี่คือประโยชน์ของ Foreign Key ที่จะเรียนในบทที่ 8 · ต้องลบคำร้องก่อน หรือย้ายคำร้องไปให้คนอื่นก่อน

สรุปบทที่ 6
  • INSERT · UPDATE · DELETE ตรงกับ POST · PUT · DELETE ที่ทำใน API
  • ไม่ต้องใส่คอลัมน์ที่มี DEFAULT — ฐานข้อมูลเติมให้เอง
  • ลืม WHERE = กระทบทุกแถว และย้อนกลับไม่ได้
  • นิสัยที่ต้องสร้าง — SELECT ดูก่อนเสมอ แล้วค่อยเปลี่ยนเป็น UPDATE/DELETE

บทที่ 7 · กลุ่มที่ 3

ทำไมต้องแยกตาราง

เวลาอ่าน ≈ 11 นาที · บทที่สำคัญที่สุดของสัปดาห์

ถึงตอนนี้เรามีตาราง requests ตารางเดียวก็ใช้งานได้แล้ว — แล้วทำไมต้องแยก users ออกมาอีก

🟩 ทายก่อนอ่านต่อ

สมมติเก็บแบบตารางเดียว โดยมีคอลัมน์ requester_name อยู่ในทุกคำร้อง
ถ้าสมชายแจ้งคำร้องไว้ 20 รายการ แล้ววันหนึ่งเขาเปลี่ยนชื่อ — ต้องแก้กี่ที่

เฉลย

20 ที่ — และถ้าแก้ไม่ครบ ฐานข้อมูลจะมีทั้งชื่อเก่าและชื่อใหม่ปนกัน

เวลาค้นด้วยชื่อใหม่จะได้ไม่ครบ · เวลาทำรายงานจะเห็นเป็นคนละคน · และไม่มีใครรู้ว่าข้อมูลผิดจนกว่าจะมีคนสังเกตเห็น

7.1 ปัญหา 4 อย่างของการเก็บซ้ำ

ปัญหาเกิดอะไรขึ้น
ข้อมูลซ้ำโดยไม่จำเป็นแจ้ง 20 ครั้ง = เก็บชื่อ 20 ครั้ง
แก้ไขแล้วไม่ครบเปลี่ยนชื่อต้องไล่แก้ทุกแถว · พลาดที่เดียวข้อมูลก็ไม่ตรงกัน
เก็บข้อมูลเพิ่มไม่ได้อยากเก็บอีเมลด้วย ต้องเพิ่มคอลัมน์ในทุกแถว และซ้ำอีก
พิมพ์ผิดแล้วไม่มีใครรู้'สมชย ใจดี' ระบบยอมรับปกติ เพราะเป็นแค่ข้อความ
Update anomaly

ปัญหาที่ แก้ข้อมูลที่เดียวไม่พอ ต้องแก้หลายที่ และมีโอกาสแก้ไม่ครบ · เป็นเหตุผลหลักที่ทำให้ต้องแยกตาราง

7.2 แยกแล้วแก้ปัญหาอย่างไร

ปัญหาเดิมแยกเป็น 2 ตารางแล้ว
ข้อมูลซ้ำชื่อเก็บที่เดียวใน users · requests เก็บแค่ตัวเลข
แก้ไม่ครบแก้ที่ users แถวเดียว ทุกคำร้องเห็นชื่อใหม่ทันที
เก็บเพิ่มไม่ได้เพิ่มคอลัมน์ใน users ได้เลย ไม่กระทบ requests
พิมพ์ผิดForeign Key ปฏิเสธ รหัสที่ไม่มีอยู่จริง
🟪 เทียบกับสิ่งที่เราทำในโค้ดมาแล้ว

นี่คือหลักการเดียวกับที่เราทำใน React และ Express — อย่าเขียนของเดิมซ้ำหลายที่
Week 05 เราแยก Service Layer ออกมาเพื่อไม่ให้ทุก component เขียนวิธีอ่านข้อมูลเอง · Week 07 เราแยก apiClient ออกมาเพื่อไม่ให้ทุกฟังก์ชันเขียน fetch เอง
ครั้งนี้คือการไม่เขียนข้อมูลซ้ำในฐานข้อมูล — หลักการเดียวกัน คนละที่

🟨 แล้วแยกมากเกินไปได้ไหม

ได้ · ถ้าแยกทุกอย่างเป็นตารางหมด จะต้อง JOIN หลายสิบตารางกว่าจะได้ข้อมูลชุดเดียว ซึ่งช้าและอ่านยาก
หลักการ — แยกเมื่อข้อมูลนั้นถูกใช้ซ้ำและเปลี่ยนแปลงได้ · location ไม่ได้แยกเพราะแต่ละคำร้องเป็นคนละที่กัน ไม่ได้ใช้ซ้ำ

สรุปบทที่ 7
  • เก็บชื่อซ้ำในทุกแถวทำให้เกิด update anomaly — แก้ที่เดียวไม่พอ
  • แยกตารางแล้วเก็บที่เดียว แก้ที่เดียว ถูกทั้งระบบ
  • หลักการเดียวกับ Service Layer และ apiClient — อย่าเขียนของเดิมซ้ำ
  • แยกเมื่อข้อมูลถูกใช้ซ้ำและเปลี่ยนแปลงได้ · ไม่ใช่แยกทุกอย่าง

บทที่ 8 · กลุ่มที่ 3

Foreign Key — เชื่อมสองตารางเข้าด้วยกัน

เวลาอ่าน ≈ 10 นาที

Foreign Key (FK)

คอลัมน์ที่ชี้ไปยัง Primary Key ของอีกตาราง · ฐานข้อมูลจะตรวจให้ว่าค่าที่ใส่มีอยู่จริงในตารางปลายทาง

CREATE TABLE requests (
  id            TEXT PRIMARY KEY,
  requester_id  INTEGER NOT NULL,
  ...
  FOREIGN KEY (requester_id) REFERENCES users(id)
);

อ่านว่า — "คอลัมน์ requester_id ในตารางนี้ ต้องเป็นค่าที่มีอยู่ในคอลัมน์ id ของตาราง users"

8.1 Foreign Key ป้องกันอะไรได้บ้าง

ถ้าลองทำฐานข้อมูลจะ
ใส่คำร้องที่ requester_id = 999 (ไม่มีคนนี้)ปฏิเสธ — FOREIGN KEY constraint failed
ลบผู้ใช้ที่ยังมีคำร้องค้างอยู่ปฏิเสธ — ไม่ให้เกิดข้อมูลกำพร้า
ลบผู้ใช้ที่ไม่มีคำร้องเลยอนุญาต
🟥 ต้องเปิดการตรวจก่อน ไม่งั้นไม่ทำงาน 
PRAGMA foreign_keys = ON;

SQLite ปิดการตรวจ Foreign Key ไว้เป็นค่าเริ่มต้น · ถ้าไม่เปิด คำสั่งที่ควรถูกปฏิเสธจะผ่านหมด และคุณจะไม่รู้เลยว่าข้อมูลเสียแล้ว
ต้องสั่งทุกครั้งที่เปิดฐานข้อมูล — ไม่ใช่สั่งครั้งเดียวแล้วจำไปตลอด

8.2 ทำไมชี้ด้วยตัวเลข ไม่ชี้ด้วยชื่อ

🟩 ทายก่อนอ่านต่อ

ทำไมไม่ให้ requests เก็บ requester_name แล้วชี้ไปที่ users.name แทน — จะได้อ่านง่ายกว่า

เฉลย

เพราะ ชื่อเปลี่ยนได้ · ถ้าสมชายเปลี่ยนชื่อ ค่าใน users.name เปลี่ยน แต่ requests.requester_name ยังเป็นชื่อเก่า — ความเชื่อมโยงขาดทันที

แต่ users.id เป็นตัวเลขที่ไม่มีวันเปลี่ยน · เปลี่ยนชื่อกี่ครั้งก็ยังเป็นคนเดิม id เดิม · นี่คือเหตุผลที่บทที่ 2 บอกว่าอย่าเลือกสิ่งที่แก้ได้มาเป็น Primary Key

8.3 ความสัมพันธ์แบบหนึ่งต่อหลาย

One-to-many (หนึ่งต่อหลาย)

ผู้ใช้ 1 คนแจ้งคำร้องได้หลายรายการ · แต่คำร้อง 1 รายการมีผู้แจ้งคนเดียว

วิธีสังเกตว่า FK ควรอยู่ตารางไหน — อยู่ฝั่ง "หลาย" เสมอ

ถ้าเก็บกลับด้านจะเกิดอะไร
ให้ users เก็บ request_idเก็บได้แค่คำร้องเดียวต่อคน · ถ้าแจ้ง 20 รายการต้องมี 20 คอลัมน์
ให้ requests เก็บ requester_id ✓แต่ละคำร้องชี้ไปคนเดียว — พอดีกับความจริง
สรุปบทที่ 8
  • Foreign Key ชี้ไปยัง Primary Key ของอีกตาราง · ฐานข้อมูลตรวจให้ว่ามีอยู่จริง
  • ป้องกันทั้งใส่ค่าที่ไม่มีจริง และลบข้อมูลที่ยังมีคนอ้างถึง
  • ต้องสั่ง PRAGMA foreign_keys = ON ทุกครั้งที่เปิดฐานข้อมูล
  • ชี้ด้วย id ที่ไม่เปลี่ยน ไม่ใช่ชื่อที่เปลี่ยนได้
  • ในความสัมพันธ์หนึ่งต่อหลาย FK อยู่ฝั่ง "หลาย" เสมอ

บทที่ 9 · กลุ่มที่ 3

JOIN — ดึงข้อมูลจากสองตารางมารวมกัน

เวลาอ่าน ≈ 12 นาที

เราแยกข้อมูลออกเป็น 2 ตารางเพื่อไม่ให้ซ้ำ — แต่เวลาใช้งานจริง ต้องการเห็นพร้อมกัน

9.1 รูปแบบของ JOIN

SELECT r.id,
       u.name AS requesterName,
       r.status
FROM requests r
JOIN users u ON u.id = r.requester_id;

อ่านทีละบรรทัด

บรรทัดหมายความว่า
FROM requests rเริ่มจากตาราง requests · ตั้งชื่อย่อว่า r
JOIN users uเอาตาราง users มาต่อ · ตั้งชื่อย่อว่า u
ON u.id = r.requester_idเชื่อมตรงไหน — แถวที่ users.id ตรงกับ requests.requester_id
🟨 ชื่อย่อ (alias) ช่วยให้อ่านง่าย

requests r ทำให้เขียน r.id แทน requests.id ได้
จำเป็นเมื่อสองตารางมีคอลัมน์ชื่อเหมือนกัน — ทั้งสองตารางมี id ถ้าไม่ระบุว่าเป็น r.id หรือ u.id ฐานข้อมูลจะไม่รู้ว่าหมายถึงอันไหน

9.2 ผลลัพธ์ที่ได้

REQ-002สุภาวดี รักเรียนin-progress

ผลลัพธ์นี้ หน้าตาเหมือนข้อมูลที่ API ส่งให้ React ตอน Week 07 ทุกประการ

{ "id": "REQ-001", "requesterName": "สมชาย ใจดี", "status": "pending" }

ในฐานข้อมูลแยก 2 ตารางเพื่อไม่ให้ข้อมูลซ้ำ · ตอนส่งออกรวมกลับเป็นก้อนเดียวเพื่อให้ฝั่งผู้ใช้ทำงานง่าย
นี่คือหน้าที่ของชั้น service — สัปดาห์หน้าเราจะเปลี่ยน requestService.js ให้ดึงจากฐานข้อมูลนี้ โดย frontend ไม่ต้องแก้อะไรเลย

9.3 JOIN ร่วมกับ WHERE

SELECT r.id, u.name, r.details
FROM requests r
JOIN users u ON u.id = r.requester_id
WHERE u.department = 'วิศวกรรมซอฟต์แวร์'
ORDER BY r.id;

สังเกตว่ากรองด้วยคอลัมน์จากตาราง users ได้ ทั้งที่เริ่มจาก requests — เพราะ JOIN ทำให้ทั้งสองตารางกลายเป็นตารางเดียวชั่วคราว

9.4 JOIN กับ LEFT JOIN

แบบคืนอะไรใช้เมื่อ
JOINเฉพาะแถวที่จับคู่ได้ทั้งสองฝั่งต้องการเฉพาะคำร้องที่มีผู้แจ้งอยู่จริง
LEFT JOINทุกแถวจากตารางซ้าย แม้จับคู่ไม่ได้ (ฝั่งขวาเป็น NULL)ต้องการรายชื่อผู้ใช้ทุกคน รวมคนที่ยังไม่เคยแจ้ง
-- ใครแจ้งกี่รายการ — คนที่ยังไม่เคยแจ้งต้องติดมาด้วย จึงต้องใช้ LEFT JOIN
SELECT u.name, COUNT(r.id) AS total
FROM users u
LEFT JOIN requests r ON r.requester_id = u.id
GROUP BY u.id
ORDER BY total DESC;
🟥 กับดักของ JOIN — ลืม ON

ถ้าเขียน FROM requests, users โดยไม่บอกเงื่อนไขการเชื่อม จะได้ ทุกแถวจับคู่กับทุกแถว
requests 5 แถว × users 4 แถว = 20 แถว ที่ไม่มีความหมายอะไรเลย · และไม่มี error บอก
ถ้าผลลัพธ์ออกมาเยอะผิดปกติ ให้ตรวจ ON เป็นอันดับแรก

9.5 JOIN จับคู่ทีละแถวอย่างไร

สรุปบทที่ 9
  • JOIN ... ON รวมสองตารางเป็นตารางเดียวชั่วคราว
  • ใช้ alias (r, u) เพื่อระบุว่าคอลัมน์มาจากตารางไหน
  • กรองด้วยคอลัมน์จากตารางไหนก็ได้หลัง JOIN แล้ว
  • LEFT JOIN เก็บแถวที่จับคู่ไม่ได้ไว้ด้วย
  • JOIN สร้างข้อมูลซ้ำตอนอ่าน ไม่ใช่ตอนเก็บ — นี่คือประโยชน์ของการแยกตาราง
  • ผลของ JOIN = รูปแบบที่ API ส่งให้ React — เป็นสะพานไปสัปดาห์หน้า

10.2 schema.sql ต้องรันซ้ำได้

-- ลบของเดิมก่อน เพื่อให้รันซ้ำได้โดยไม่ error
-- ⚠ ลบ requests ก่อน เพราะมันอ้างถึง users
DROP TABLE IF EXISTS requests;
DROP TABLE IF EXISTS users;

CREATE TABLE users ( ... );
CREATE TABLE requests ( ... );

INSERT INTO users ... ;
INSERT INTO requests ... ;
🟥 ลำดับการลบสำคัญ

ต้องลบ requests ก่อน users เสมอ
เพราะ requests มี Foreign Key ชี้ไป users — ถ้าลบ users ก่อน จะเหลือคำร้องที่ชี้ไปหาตารางที่ไม่มีแล้ว
กฎง่าย ๆ — ลบฝั่งที่ "ชี้ไปหา" ก่อน แล้วค่อยลบฝั่งที่ "ถูกชี้"

10.3 ทดสอบว่า constraint ทำงานจริง (CP25)

การเขียน constraint ไว้ไม่ได้แปลว่ามันทำงาน — ต้องพิสูจน์ด้วยการลองทำผิดแล้วดูว่าถูกปฏิเสธจริงไหม

ทดสอบต้องได้ error
ใส่คำร้องที่ requester_id ไม่มีจริงFOREIGN KEY constraint failed
ใส่ status นอกรายการที่กำหนดCHECK constraint failed
ใส่อีเมลซ้ำกับคนที่มีอยู่UNIQUE constraint failed
ใส่คำร้องที่ id ซ้ำUNIQUE constraint failed
ใส่คำร้องโดยไม่ระบุ locationNOT NULL constraint failed
🟨 การทดสอบที่ดีคือการทดสอบว่ามันปฏิเสธได้

เหมือนที่เราทำใน Week 06 — ทดสอบว่า API ตอบ 400 เมื่อข้อมูลไม่ครบ ไม่ใช่แค่ทดสอบว่าตอบ 201 เมื่อข้อมูลถูก
ระบบที่ดีต้องปฏิเสธสิ่งที่ผิดได้ ไม่ใช่แค่รับสิ่งที่ถูก

10.5 แผนผังฐานข้อมูลที่ต้องส่ง

ภาพนี้คือข้อกำหนดของงานสัปดาห์นี้ — ฐานข้อมูลที่คุณสร้างต้องมีหน้าตาแบบนี้

🟥 ชื่อคอลัมน์ต้องตรงทุกตัวอักษร

สัปดาห์ที่ 10 จะเขียนโค้ดที่อ้างชื่อคอลัมน์เหล่านี้ตรง ๆ เช่น r.request_type AS requestType
ถ้าตั้งชื่อเป็น requestType หรือ type แทน โค้ดสัปดาห์หน้าจะใช้ไม่ได้ทันที
ในงานจริงก็เป็นแบบนี้ — เมื่อตกลง schema แล้ว การเปลี่ยนชื่อคอลัมน์กระทบทุกคนที่ใช้ฐานข้อมูลนั้น

🟨 ทำไมฝั่งฐานข้อมูลใช้ request_type แต่ API ใช้ requestType

เป็นธรรมเนียมคนละฝั่ง — ฐานข้อมูลนิยมขีดล่าง (snake_case) · JavaScript นิยมตัวใหญ่กลางคำ (camelCase)
เราไม่ฝืนธรรมเนียมทั้งสองฝั่ง แต่ใช้ AS แปลงตอนดึงออกมา — สัปดาห์หน้าจะได้ใช้จริง
