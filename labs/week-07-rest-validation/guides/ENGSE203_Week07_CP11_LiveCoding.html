TODO W07-F2 · We do
export async function apiFetch(path, options = {}) {
  let response;
  try {
    response = await fetch(`${BASE_URL}${path}`, {
      headers: { 'Content-Type': 'application/json', ...options.headers },
      ...options,
    });
  } catch {
    // ① ต่อเซิร์ฟเวอร์ไม่ได้เลย — API ไม่ได้เปิด หรือเน็ตหลุด
    throw new ApiError('ติดต่อเซิร์ฟเวอร์ไม่ได้ — ตรวจว่าเปิด API ที่พอร์ต 3001 แล้วหรือยัง', 0);
  }

  if (!response.ok) {
    // ② เซิร์ฟเวอร์ตอบ แต่เป็น 4xx/5xx
    throw new ApiError(await parseError(response), response.status);
  }

  if (response.status === 204) return null;   // ③ ไม่มี body
  return response.json();                       // ④ ปกติ
}
⚠ กับดักที่เจอมาแล้วตั้งแต่ Week 05 — ย้ำอีกครั้ง

fetch ไม่โยน error เมื่อได้ 404 หรือ 500 — มันจะ resolve สำเร็จ ต้องเช็ค response.ok เอง

fetch โยน error เฉพาะตอนต่อเซิร์ฟเวอร์ไม่ได้เลย — นี่คือเหตุผลที่ต้องมี try/catch รอบ fetch แยกจากการเช็ค response.ok

อธิบายทำไมต้องเก็บ status ไว้ใน error "เพราะคนเรียกต้องตัดสินใจต่างกันตาม status — 404 อาจไม่ใช่ความผิดพลาด (แค่ไม่มีคำร้องรหัสนั้น) แต่ 500 คือระบบพังจริง
ถ้าไม่เก็บ status ไว้ จะแยกไม่ออก · เดี๋ยวเราจะใช้ตอน CP11 ส่วนที่สอง"

จุดที่ 1 · getRequestById ต้องแปลง 404 เป็น null

export async function getRequestById(requestId) {
  try {
    return await apiFetch(`/api/requests/${encodeURIComponent(requestId)}`);
  } catch (error) {
    // 404 ไม่ใช่ความผิดพลาดของระบบ — แปลว่าไม่มีคำร้องรหัสนี้
    if (error instanceof ApiError && error.status === 404) return null;
    throw error;   // error อื่นปล่อยผ่านไปให้หน้าจอจัดการ
  }
}
อธิบายว่าทำไม "Week 05 ฟังก์ชันนี้คืน null เมื่อไม่พบ และ RequestDetailPage เขียนไว้ว่า if (!request) แสดงว่าไม่พบ
ถ้าเราปล่อยให้โยน error หน้าจอจะขึ้น 'เกิดข้อผิดพลาด' แทนที่จะขึ้น 'ไม่พบคำร้อง' — ซึ่งสื่อสารผิดกับผู้ใช้"

export async function deleteRequest(requestId) {
  await apiFetch(`/api/requests/${encodeURIComponent(requestId)}`, { method: 'DELETE' });
  // คืนรายการล่าสุดจากเซิร์ฟเวอร์ เพื่อให้หน้าจอตรงกับข้อมูลจริงเสมอ
  return getRequests();
}
