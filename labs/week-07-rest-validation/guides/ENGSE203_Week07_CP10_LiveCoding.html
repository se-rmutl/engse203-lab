config.js — รวมค่าตั้งค่าไว้ที่เดียว

api/src/config.js · TODO W07-CFG
อธิบายก่อนพิมพ์ "ก่อนจะเปิด CORS เราต้องรู้ก่อนว่าจะอนุญาตให้ใครเรียก — ซึ่งค่านี้ไม่ควรฝังไว้ในโค้ด เพราะเครื่องแต่ละคนอาจใช้พอร์ตต่างกัน และตอนขึ้นเซิร์ฟเวอร์จริงก็คนละค่าอีก" 
import 'dotenv/config';

export const config = {
  port: Number(process.env.PORT ?? 3001),
  corsOrigin: process.env.CORS_ORIGIN ?? 'http://localhost:5173',
  nodeEnv: process.env.NODE_ENV ?? 'development',
  get isProduction() {
    return this.nodeEnv === 'production';
  },
};
⚠ ชี้จุดนี้ให้ดี — process.env ได้ string เสมอ process.env.PORT ได้ "3001" เป็นข้อความ ไม่ใช่ตัวเลข
ต้อง Number() เอง · ถ้าลืมจะทำงานผิดในบางที่โดยไม่มี error บอก ซึ่งหาสาเหตุยากมาก
ทำไมต้องรวมไว้ไฟล์เดียว "เปิดไฟล์นี้ไฟล์เดียวก็รู้ทันทีว่าแอปนี้ต้องการค่าอะไรบ้างถึงจะรันได้
ถ้าปล่อยให้อ่าน process.env กระจายทั่วโปรเจกต์ คนใหม่ที่มาทำต่อจะไม่รู้เลยว่าต้องตั้งค่าอะไร — เป็นหลักการเดียวกับ Service Layer ที่รวมสิ่งที่เปลี่ยนบ่อยไว้ที่เดียว"

และดูไฟล์ .env ที่ให้มาแล้ว

# api/.env
PORT=3001
CORS_ORIGIN=http://localhost:5173
NODE_ENV=development
②

เปิด CORS

api/src/app.js · TODO W07-A1
import cors from 'cors';
import { config } from './config.js';

export function createApp() {
  const app = express();

  app.use(cors({ origin: config.corsOrigin }));   ← บนสุด!

  app.use(express.json());
  // route ทั้งหมดอยู่ข้างล่างนี้
}
อธิบายว่าบรรทัดนี้ทำอะไร "บรรทัดเดียวนี้บอกเบราว์เซอร์ว่า 'เว็บที่อยู่ที่ 5173 เรียกฉันได้นะ'
โดยแนบ header Access-Control-Allow-Origin ไปกับทุก response"
⚠ ต้องอยู่บนสุด — เหตุผล สำหรับ POST/PUT/DELETE เบราว์เซอร์จะส่ง preflight (OPTIONS) มาก่อน
ถ้า cors อยู่หลัง route คำขอ OPTIONS จะไปไม่ถึง เพราะ route ตอบไปก่อนแล้ว · เป็นหลักการเดียวกับ middleware ทุกตัวใน Week 06

ทดสอบ — refresh แล้วดู

8 นาที
ให้ทุกคน refresh พร้อมกัน "กลับไปที่หน้า Dashboard แล้วกด refresh ครับ — ดู Console"
ตรวจอะไรต้องได้
Consoleไม่มี CORS error แล้ว
Network → Response Headersมี Access-Control-Allow-Origin: http://localhost:5173
หน้าจอข้อมูลจาก API ขึ้นมาแล้ว (อาจแสดงไม่สวยเพราะยังไม่ได้ทำ CP11)
ให้ยกมือรายงาน "ใครที่ Console ไม่มีสีแดงแล้วยกมือครับ" — รอให้เกือบทั้งห้องยกมือก่อนไปต่อ
สาธิต preflight ให้ดู ลองกดสร้างคำร้องใหม่ แล้วดู Network tab → จะเห็น 2 บรรทัด
OPTIONS ตามด้วย POST
"อันแรกคือเบราว์เซอร์ส่งไปถามก่อนว่า POST ได้ไหม — เราไม่ได้เขียนเอง เบราว์เซอร์ทำให้"
สาธิตว่า env ทำงานจริง (ถ้ามีเวลา) แก้ PORT=3002 ใน .env → รีสตาร์ท → เห็น API ย้ายพอร์ตจริง
แล้วเปลี่ยนกลับเป็น 3001 ก่อนไปต่อ ไม่งั้นจะงงกันทั้งห้อง

กับดักของช่วงนี้

พบบ่อยที่สุด วาง cors ผิดที่

✕ ผิด 
app.use('/api/requests', routes);
app.use(cors(...));   ← สายไป
✓ ถูก 
app.use(cors(...));   ← บนสุด
app.use('/api/requests', routes);

GET อาจพอผ่าน แต่ POST จะพัง เพราะ preflight ไปไม่ถึง
