service — จัดการข้อมูล

src/services/requestService.js · TODO W06-S1
export function findAll({ status } = {}) {
  if (!status) return structuredClone(requests);
  return structuredClone(requests.filter((r) => r.status === status));
}
ชี้ 2 จุด หนึ่ง — structuredClone() คืนสำเนา เพื่อไม่ให้ข้างนอกแก้ข้อมูลต้นฉบับได้ · หลักการเดียวกับที่ใช้ใน Week 05
สอง — ส่วน status คือ ⭐ Challenge ทำตอนนี้เลยก็ได้ ไม่เสียหาย
ย้ำเรื่อง service ไม่รู้จัก HTTP "สังเกตว่าในไฟล์นี้ไม่มีคำว่า req หรือ res เลย — service ไม่รู้ว่ามันถูกเรียกจาก HTTP หรือจากที่ไหน
นี่คือเหตุผลที่หน่วย 4 เราจะเปลี่ยนไปใช้ database ได้โดยแก้แค่ไฟล์นี้"

controller — ตัดสิน status

src/controllers/requestController.js · TODO W06-C1
export function listRequests(req, res) {
  const { status } = req.query;
  res.status(200).json(service.findAll({ status }));
}
controller ต่างจาก service อย่างไร controller รู้จัก req/res และเป็นคนตัดสินว่าจะตอบ status อะไร · แต่ไม่จัดการข้อมูลเอง — ส่งต่อให้ service ทำ
อธิบาย req.query "req.query คือค่าที่มาหลังเครื่องหมาย ? ใน URL · ถ้ายิง /api/requests?status=pending ก็จะได้ { status: 'pending' }
ตอนนี้ยังไม่มีใครส่งมา จะเป็น undefined ซึ่ง service รับมือได้อยู่แล้ว"

กับดักสำคัญ — ลำดับ route

ต้องเตือนก่อนเกิด route เจาะจงต้องมาก่อน

✕ ผิด 
router.get('/:id', ...)   ← ดักทุกอย่าง
router.get('/', ...)
✓ ถูก 
router.get('/', ...)      ← เจาะจงก่อน
router.get('/:id', ...)

Express เลือก route ที่เขียนไว้ก่อน · ถ้า :id อยู่บน คำขอ /api/requests/new จะถูกจับเป็น id = 'new'
