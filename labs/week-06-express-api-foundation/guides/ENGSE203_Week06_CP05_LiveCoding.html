TODO W06-C5 — controller ตอบ 204

export function deleteRequest(req, res) {
  const removed = service.remove(req.params.id);
  if (!removed) {
    return res.status(404).json({ error: `ไม่พบคำร้องรหัส ${req.params.id}` });
  }
  res.status(204).end();
}
ทำไม 204 และทำไม .end() "204 No Content แปลว่าสำเร็จแต่ไม่มีข้อมูลส่งกลับ — เพราะลบไปแล้ว ไม่มีอะไรจะส่ง
จึงใช้ .end() ไม่ใช่ .json() · ถ้าใช้ .json() จะขัดกับความหมายของ 204"
ถามห้องระหว่างรอ "ทำไมเราไม่ตอบ 200 พร้อมข้อความว่า 'ลบสำเร็จ' ล่ะครับ"
คำตอบ — ทำได้ แต่ 204 สื่อชัดกว่าว่าไม่มีอะไรให้อ่าน · และเป็นมาตรฐานที่คนทั้งโลกใช้
