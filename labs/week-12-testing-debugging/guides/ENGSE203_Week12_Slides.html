บทที่ 2 / 5 · Unit test · Vitest หลักการ

unit test = เรียกฟังก์ชันเดียว แล้วเทียบผล

ทดสอบได้ดีที่สุดกับ pure function

รับข้อมูลเข้า → คืนผลลัพธ์ · ไม่แตะอะไรภายนอก (req/res · DB · ไฟล์) → ใส่ค่าเดิมได้ผลเดิมทุกครั้ง

AAA — โครงของ test ทุกข้อ

Arrange เตรียมข้อมูล → Act เรียกฟังก์ชัน → Assert ตรวจผล

1 test = 1 เรื่อง

ชื่อ test บอกกรณีและผลที่คาดหวัง เช่น '79 คะแนน → B' · test แต่ละข้อไม่พึ่งกัน

// ตัวอย่างที่ใช้ทั้งบทนี้ — src/utils/grade.js
export function gradeOf(score) {
  if (typeof score !== 'number' || score < 0 || score > 100) {
    throw new Error('คะแนนต้องอยู่ระหว่าง 0–100');
  }
  if (score >= 80) return 'A';
  if (score >= 70) return 'B';
  if (score >= 60) return 'C';
  if (score >= 50) return 'D';
  return 'F';
}

ถ้าโค้ดผูกกับ req/res หรือ DB จะ unit test ยาก → แยกกฎออกมาเป็น pure function แล้ว test ตัวนั้น

บทที่ 2 / 5 · Unit test · Vitest describe · test · expect

โครงของไฟล์ test — 3 คำสั่งหลัก

// tests/unit/grade.test.js
import { describe, test, expect } from 'vitest';
import { gradeOf } from '../../src/utils/grade.js';

describe('gradeOf', () => {          // กลุ่มของ test
  test('85 คะแนน → A', () => {        // 1 กรณี
    // Arrange
    const score = 85;
    // Act
    const result = gradeOf(score);
    // Assert
    expect(result).toBe('A');
  });

  test('45 คะแนน → F', () => {
    expect(gradeOf(45)).toBe('F');   // สั้น ๆ ได้ถ้าอ่านง่าย
  });
});

describe(ชื่อ, fn)

จัดกลุ่ม test ของฟังก์ชันเดียวกัน · รายงานอ่านง่าย

test(ชื่อ, fn)

1 กรณี · ชื่อ = กรณี → ผลที่คาดหวัง · (it คือชื่อเดียวกัน)

expect(ค่าจริง).matcher(ค่าที่คาด)

ไม่ตรง → test fail พร้อมบอก Expected / Received

บทที่ 2 / 5 · Unit test · Vitest expect(...).matcher()

matcher ที่ใช้บ่อย — พอสำหรับแลปทั้งหมด

matcherใช้ตรวจตัวอย่าง
toBe(x)ค่าเดี่ยวเท่ากันแบบ === (ตัวเลข · ข้อความ · true/false)expect(gradeOf(80)).toBe('A')
toEqual(x)เนื้อหาของ object / array เท่ากันexpect(errors).toEqual([])
toHaveLength(n)array / ข้อความมีกี่ตัวexpect(errors).toHaveLength(1)
toContain(x)มีสมาชิกนี้อยู่ใน array / ข้อความexpect(['A','B']).toContain('A')
toHaveProperty(k)object มี key นี้expect(body).toHaveProperty('id')
toThrow()ฟังก์ชัน throw error · ต้องห่อด้วย () =>expect(() => gradeOf(120)).toThrow()
.notกลับความหมายexpect(body).not.toHaveProperty('password')

พลาดบ่อย: expect([]).toBe([]) → fail เพราะเป็นคนละ array — เทียบ object / array ใช้ toEqual

export function createApp() {
  const app = express();
  // ลืม app.use(express.json())  ← req.body = undefined

// frontend/src/utils/requestSummary.js
export function summarizeRequests(requests) {
  const count = (status) =>
    requests.filter((r) => r.status === status).length;
  return {
    total: requests.length,
    pending: count('pending'),
    inProgress: count('in progress'),  // ช่องว่าง ≠ ขีด
    completed: count('completed'),
  };
}
