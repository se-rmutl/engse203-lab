export const MAX_NAME = 100;  export const MAX_LOCATION = 100;  export const MAX_DETAILS = 1000;
function checkText(value, label, { min = 1, max }) {          // แนวทางในเฉลย
  if (typeof value !== 'string') return `${label}ต้องเป็นข้อความ`;  // ① ชนิดก่อน
  const length = value.trim().length;
  if (length < min) return `${label}ต้องมีอย่างน้อย ${min} ตัวอักษร`;
  if (length > max) return `${label}ต้องไม่เกิน ${max} ตัวอักษร`;  // ② ค่าสูงสุด
  return null;
}
// app.js — ③ ด่านแรกสุด: ทั้งก้อนเกิน 10kb → 413 "ข้อมูลที่ส่งมามีขนาดใหญ่เกินกำหนด"

Authentication vs Authorization

สองคำถามที่ต่างกัน ได้ status code ต่างกัน — 401 กับ 403

🏫 เรียนในห้อง≈ 5 นาที
บทที่ 3 / 9 · authN vs authZ สองคำถาม สองด่าน

"คุณคือใคร" → 401 · "ทำสิ่งนี้ได้ไหม" → 403

Authentication · คุณคือใคร

ไม่มี token · token ปลอม · หมดอายุ → 401 Unauthorized
แปลจริง ๆ ว่า "ยังไม่รู้ว่าคุณเป็นใคร" — เข้าสู่ระบบใหม่แล้วแก้ได้

Authorization · ทำได้ไหม

รู้ว่าเป็นใคร แต่ role ไม่ใช่ staff → 403 Forbidden
เข้าสู่ระบบกี่รอบก็ไม่ช่วย — ต้องมีสิทธิ์

GET ✓ ทุกคนPOST ✓ ทุกคน (ใครก็ส่งคำร้องได้) PUT 🔒 เจ้าหน้าที่ DELETE 🔒 เจ้าหน้าที่

เหมือนยามหน้าอาคาร — ไม่มีบัตร/บัตรปลอม = 401 ไม่รู้ว่าเป็นใคร · มีบัตรนักศึกษาจริงแต่จะเข้าห้องควบคุมไฟ = 403

ผู้แจ้งทั่วไปมี token ถูกต้อง (role: 'requester') แล้วกดลบคำร้อง — API ควรตอบอะไร

บทที่ 3 / 9 · authN vs authZ

จบบทที่ 3 — ตัวตน vs สิทธิ์

✓ สิ่งที่ต้องจำจากบทนี้

  • Authentication = คุณคือใคร · ไม่รู้ (ไม่มี/ปลอม/หมดอายุ) → 401
  • Authorization = ทำสิ่งนี้ได้ไหม · รู้ว่าใครแต่ไม่มีสิทธิ์ → 403
  • นโยบาย: GET และ POST เปิดให้ทุกคน · PUT และ DELETE เฉพาะเจ้าหน้าที่
  • ต้องรู้ "ใคร" ก่อนถึงจะตัดสิน "ได้ไหม" — ลำดับด่านสำคัญ
ต่อไป → บทที่ 4 — จะรู้ว่า "ใคร" ต้องตรวจรหัสผ่าน · แล้วเก็บรหัสผ่านอย่างไรให้ปลอดภัย

❌ เก็บตรง ๆ

DB หลุด = รหัสทุกคนหลุดทันที (คนใช้รหัสซ้ำหลายเว็บ)

❌ sha256

เร็วเกิน — เดาได้หลายล้านครั้งต่อวินาที

⚠ bcrypt

ดี แต่เป็น native module ต้อง compile — มักพังบน Windows

✅ scrypt + salt

ช้าโดยตั้งใจ · มากับ Node (node:crypto) ไม่ต้องติดตั้ง

เก็บเป็น scrypt$salt$hash · เทียบด้วย timingSafeEqual

import { randomBytes, scryptSync, timingSafeEqual } from 'node:crypto';
export function hashPassword(plain) {
  const salt = randomBytes(16).toString('hex');                 // 32 ตัวอักษร
  const hash = scryptSync(plain, salt, 64).toString('hex');       // 128 ตัวอักษร
  return `scrypt$${salt}$${hash}`;
}
export function verifyPassword(plain, stored) {
  const [scheme, salt, hashHex] = String(stored ?? '').split('$');
  if (scheme !== 'scrypt' || !salt || !hashHex) return false;      // ผิดรูปแบบ → false ไม่โยน error
  const expected = Buffer.from(hashHex, 'hex');
  const actual = scryptSync(String(plain), salt, expected.length);  // salt เดิม → hash ใหม่
  return actual.length === expected.length && timingSafeEqual(actual, expected);
}
timingSafeEqual

เทียบครบทุกไบต์ ใช้เวลาเท่ากันเสมอ · === หยุดทันทีที่เจอตัวแรกที่ต่าง → วัดเวลาตอบแล้วเดาทีละไบต์ได้

salt ไม่ใช่ความลับ — เก็บคู่กับ hash ได้ เพราะหน้าที่ของมันคือทำให้ hash ไม่ซ้ำกัน · ⚠ รูปแบบต้องตรงเป๊ะ ไม่งั้น hash ของ staff1234 ใน schema.sql จะตรวจไม่ผ่าน

export function login(email, password) {
  const user = findUserByEmail(email);
  if (!user || user.role !== 'staff' || !verifyPassword(password, user.passwordHash)) {
    return null;                                   // ไม่มีอีเมล กับ รหัสผิด → ผลเดียวกัน
  }
  const payload = { sub: String(user.id), name: user.name, role: user.role };  // ห้ามใส่รหัสผ่าน/hash
  const token = jwt.sign(payload, config.jwtSecret, { expiresIn: config.jwtExpiresIn });  // '2h'
  return { token, user: { id: user.id, name: user.name, role: user.role } };
}
export function verifyToken(token) { return jwt.verify(token, config.jwtSecret); }  // ปลอม/หมดอายุ → throw

  • payload เป็นแค่ base64url — อ่านได้ทุกคน ห้ามใส่รหัสผ่าน
  • แก้ payload แล้ว signature ไม่ตรง → jwt.verify ไม่ผ่าน → 401
  • stateless — server ไม่ต้องจำ session · หมดอายุเองด้วย exp (expiresIn: '2h')
  • ไม่มีอีเมล กับ รหัสผิด → 401 "อีเมลหรือรหัสผ่านไม่ถูกต้อง" เหมือนกัน

export function authenticate(req, res, next) {
  const [scheme, token] = (req.get('Authorization') ?? '').split(' ');
  if (scheme !== 'Bearer' || !token) {
    res.set('WWW-Authenticate', 'Bearer');
    return res.status(401).json({ error: 'ต้องเข้าสู่ระบบก่อน' });
  }
  try { req.user = verifyToken(token); next(); }          // แนบผู้ใช้ไว้ให้ด่านถัดไป
  catch { res.set('WWW-Authenticate', 'Bearer error="invalid_token"');
          return res.status(401).json({ error: 'token ไม่ถูกต้องหรือหมดอายุ กรุณาเข้าสู่ระบบใหม่' }); }
}
export function requireRole(role) {                               // คืน middleware — ตั้งค่าได้
  return (req, res, next) => req.user?.role !== role
    ? res.status(403).json({ error: 'ไม่มีสิทธิ์ทำรายการนี้' }) : next();
}

function readJwtSecret() {
  const secret = process.env.JWT_SECRET;
  if (secret) return secret;
