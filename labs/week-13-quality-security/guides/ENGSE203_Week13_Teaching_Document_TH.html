🟨 "ข้อมูลจากภายนอก" มีมากกว่า body

🟥 "ซ่อนปุ่มแล้ว = ปลอดภัย" ไม่จริง

บทที่ 8 เราจะซ่อนปุ่ม "ลบ" จากผู้ที่ไม่ใช่เจ้าหน้าที่ · นั่นคือ UX ไม่ใช่ความปลอดภัย — คนที่รู้ URL ยังยิง DELETE /api/requests/REQ-001 ได้ · สิ่งที่กันจริงคือ middleware ฝั่ง server (บทที่ 6)

สรุปบทที่ 1
  • ทุกอย่างที่มาจากเบราว์เซอร์หรือเครื่องมือยิง API อยู่นอกขอบเขตความเชื่อใจ — ใครก็แก้ได้
  • ตรวจที่ frontend = UX · ตรวจที่ server = security · ต้องมีทั้งคู่ แต่ตัวที่ขาดไม่ได้คือ server
  • ข้อมูลภายนอกรวม body · params · query · header (รวม token)
  • ซ่อนปุ่มไม่ได้กันอะไร — server ต้องบังคับใช้กฎเอง

2.2 เขียนใหม่ — ตรวจชนิดก่อน แล้วค่อยตรวจความยาว

// api/src/validators/requestValidator.js (หลัง CP48)
export const MIN_NAME = 2;
export const MAX_NAME = 100;
export const MAX_LOCATION = 100;
export const MIN_DETAILS = 10;
export const MAX_DETAILS = 1000;

/** ตรวจช่องข้อความหนึ่งช่อง — คืนข้อความ error หรือ null */
function checkText(value, label, { min = 1, max }) {
  if (typeof value !== 'string') return `${label}ต้องเป็นข้อความ`;      // ① ชนิดก่อน
  const length = value.trim().length;
  if (length < min) return min > 1 ? `${label}ต้องมีอย่างน้อย ${min} ตัวอักษร` : `กรุณาระบุ${label}`;
  if (length > max) return `${label}ต้องไม่เกิน ${max} ตัวอักษร`;  // ② ขั้นสูงสุด
  return null;
}

export function validateRequestInput(input) {
  ...
  const errors = [
    checkText(input.requesterName, 'ชื่อผู้แจ้ง', { min: MIN_NAME, max: MAX_NAME }),
    REQUEST_TYPES.includes(input.requestType) ? null : 'ประเภทคำร้องไม่ถูกต้อง',
    checkText(input.location, 'สถานที่', { max: MAX_LOCATION }),
    checkText(input.details, 'รายละเอียด', { min: MIN_DETAILS, max: MAX_DETAILS }),
    PRIORITIES.includes(input.priority) ? null : 'ความเร่งด่วนต้องเป็น normal หรือ urgent',
  ];
  return errors.filter(Boolean);   // ตัด null ทิ้ง เหลือแต่ข้อความ error
}

Authentication กับ Authorization — 401 ต่างจาก 403

เวลาอ่าน ≈ 9 นาที · อ่านก่อนเข้าคาบ · พื้นฐานของ CP49–CP51

ตอนนี้ใครก็ลบคำร้องได้ — แค่รู้ URL · ก่อนจะเขียนโค้ดสักบรรทัด ต้องแยกให้ออกก่อนว่าคำถาม 2 ข้อนี้ต่างกัน

Authentication (การยืนยันตัวตน · authn)

ตอบคำถาม "คุณคือใคร" — พิสูจน์ด้วยสิ่งที่เจ้าของเท่านั้นรู้หรือมี เช่น รหัสผ่าน หรือ token ที่ server ออกให้

Authorization (การตรวจสิทธิ์ · authz)

ตอบคำถาม "คุณทำสิ่งนี้ได้ไหม" — เกิดหลังรู้แล้วว่าเป็นใคร · ตัดสินจากบทบาท (role) และนโยบายของระบบ

🟪 เทียบกับบัตรพนักงานกับสิทธิ์เข้าห้อง

ไม่มีบัตร หรือบัตรปลอม/หมดอายุ → รปภ. ไม่รู้ว่าคุณเป็นใคร — "กรุณาแสดงบัตร" = 401
มีบัตรจริง แต่เป็นบัตรนักศึกษา จะเข้าห้องเซิร์ฟเวอร์ → รู้ว่าเป็นใคร แต่ไม่มีสิทธิ์เข้าห้องนี้ = 403
แสดงบัตรอีกกี่ครั้งก็ได้ผลเดิม — ต้องไปขอสิทธิ์ ไม่ใช่ขอบัตรใหม่

เก็บรหัสผ่านอย่างไร — scrypt + salt

เวลาอ่าน ≈ 13 นาที · ตรงกับ CP49 · TODO W13-HASH · ช่วง 13:35–14:10

ก่อนจะให้เจ้าหน้าที่เข้าสู่ระบบได้ ต้องตัดสินใจก่อนว่าจะเก็บรหัสผ่านไว้ในฐานข้อมูลอย่างไร · ตั้งสมมติฐานไว้ก่อนว่าวันหนึ่งไฟล์ฐานข้อมูลจะหลุด (backup ตกหล่น · commit ผิดไฟล์ · server ถูกเจาะ) — การเก็บที่ดีคือการเก็บที่หลุดแล้วยังไม่เสียหายมาก

🟩 ทายก่อนอ่านต่อ

ถ้าเก็บรหัสผ่านเป็น sha256('staff1234') แทนตัวจริง — ฐานข้อมูลหลุดแล้วคนที่ได้ไฟล์ไปจะหารหัสผ่านจริงได้ไหม · ในเมื่อ hash ย้อนกลับไม่ได้

เฉลย

ได้ — สำหรับรหัสผ่านทั่วไป · ไม่ต้องย้อนกลับ hash เลย แค่เดาแล้ว hash เทียบ

  • sha256 ออกแบบมาให้เร็ว — เครื่องโน้ตบุ๊กธรรมดา hash ได้หลายแสนครั้งต่อวินาที (ทดลองจริงบนเครื่องทดสอบ: 100,000 ครั้งใช้เวลาประมาณ 0.3 วินาที) · การ์ดจอทำได้เร็วกว่านั้นมาก
  • รหัสผ่านเดียวกันได้ hash เดียวกันเสมอ — มีตาราง hash ของรหัสผ่านยอดนิยมที่คำนวณไว้ล่วงหน้าแล้ว เปิดดูได้ทันที
  • ผู้ใช้สองคนที่ใช้รหัสเดียวกัน เห็นได้ทันทีจาก hash ที่ซ้ำกัน

ต้องการ 2 อย่างที่ sha256 ไม่มี — ช้าโดยตั้งใจ และ salt สุ่ม

4.1 สามระดับของการเก็บรหัสผ่าน

① plaintext staff1234อ่านได้ทันที · ผู้ใช้ที่ใช้รหัสเดียวกับอีเมลหรือธนาคารเสียหายต่อ❌ ห้ามเด็ดขาด

ค่าสุ่มที่สร้างใหม่ทุกครั้งที่ hash แล้วเก็บไว้คู่กับ hash (ไม่ใช่ความลับ) · ทำให้รหัสผ่านเดียวกันได้ hash ต่างกัน — ตาราง hash ที่คำนวณไว้ล่วงหน้าจึงใช้ไม่ได้

4.4 โค้ดที่ทำให้ผ่าน

// api/src/utils/password.js
import { randomBytes, scryptSync, timingSafeEqual } from 'node:crypto';

const KEY_LENGTH = 64;

export function hashPassword(plain) {
  const salt = randomBytes(16).toString('hex');                 // สุ่มใหม่ทุกครั้ง
  const hash = scryptSync(plain, salt, KEY_LENGTH).toString('hex');
  return `scrypt$${salt}$${hash}`;
}

export function verifyPassword(plain, stored) {
  const [scheme, salt, hashHex] = String(stored ?? '').split('$');
  if (scheme !== 'scrypt' || !salt || !hashHex) return false;   // ผิดรูปแบบ → false ไม่ throw
  const expected = Buffer.from(hashHex, 'hex');
  const actual = scryptSync(String(plain), salt, expected.length);
  // timingSafeEqual ใช้เวลาเท่ากันไม่ว่าจะผิดตัวที่เท่าไร — กันการเดาจากเวลาที่ใช้ตอบ
  return actual.length === expected.length && timingSafeEqual(actual, expected);
}
🟩 ทายก่อนอ่านต่อ

ทำไมต้องเทียบด้วย timingSafeEqual(actual, expected) — เขียน actual.toString('hex') === hashHex ก็ได้ผลถูกเหมือนกันไม่ใช่หรือ

เฉลย

ผลถูกเหมือนกัน แต่เวลาที่ใช้ต่างกัน — === หยุดทันทีที่เจอตัวอักษรแรกที่ไม่ตรง · ค่าที่ตรงกันยาวกว่าจึงใช้เวลานานกว่าเล็กน้อย

คนที่วัดเวลาตอบได้ละเอียดพอ อาจใช้ความต่างนั้นเดาทีละส่วนได้ (เรียกว่า timing attack) · timingSafeEqual เทียบครบทุกไบต์เสมอ ใช้เวลาเท่ากันไม่ว่าจะผิดตรงไหน

ข้อควรรู้: timingSafeEqual ต้องการ Buffer ยาวเท่ากัน ไม่งั้น throw · โค้ดจึงเช็ค actual.length === expected.length ก่อน และสร้าง actual ด้วยความยาวของ hash เดิม

🟥 salt ต้องเป็น "ข้อความ hex" — ไม่ใช่ Buffer

hash ของ staff1234 ใน schema.sql สร้างด้วย scryptSync(รหัส, salt ที่เป็นข้อความ hex, 64) · ถ้าเราแปลง salt กลับเป็น Buffer ก่อน (Buffer.from(salt, 'hex')) ผลจะต่างไป (ทดลองแล้ว) → test "ตรวจ hash ของบัญชีเจ้าหน้าที่ใน schema.sql ได้" แดง และเข้าสู่ระบบด้วยบัญชีนี้ไม่ได้ · ใช้ salt เป็น string ตามที่แยกออกมาจาก split('$') ตรง ๆ

4.5 ทำไมไม่ใช้ bcrypt

bcrypt เป็นตัวเลือกยอดนิยมที่ดีเช่นกัน แต่แพ็กเกจหลักต้อง compile native module ตอน npm install ซึ่งมักพังบน Windows (ต้องมี build tools) · scrypt อยู่ใน node:crypto แล้ว — แนวคิดเดียวกับที่เลือก node:sqlite ในหน่วยที่ 4: ใช้ของที่มากับ Node ไม่ต้องติดตั้งเพิ่ม ทุกเครื่องในห้องใช้ได้เหมือนกัน

5.3 เขียน login() ใน authService

// api/src/services/authService.js
import jwt from 'jsonwebtoken';
import { config } from '../config.js';
import { findUserByEmail } from './requestService.js';
import { verifyPassword } from '../utils/password.js';

export function login(email, password) {
  const user = findUserByEmail(email);
  if (!user || user.role !== 'staff' || !verifyPassword(password, user.passwordHash)) {
    return null;          // ไม่มีอีเมล · ไม่ใช่ staff · รหัสผิด → ผลเดียวกันหมด
  }
  // payload อ่านได้ทุกคน (แค่ base64) — ใส่เฉพาะสิ่งที่ไม่ลับ ห้ามใส่รหัสผ่าน
  const payload = { sub: String(user.id), name: user.name, role: user.role };
  const token = jwt.sign(payload, config.jwtSecret, { expiresIn: config.jwtExpiresIn });  // '2h'
  return { token, user: { id: user.id, name: user.name, role: user.role } };
}

/** ตรวจ token — ถูกต้องคืน payload · ปลอม/หมดอายุ โยน error */
export function verifyToken(token) {
  return jwt.verify(token, config.jwtSecret);
}

6.1 เขียน middleware ทั้งสองตัว

// api/src/middleware/auth.js
import { verifyToken } from '../services/authService.js';

export function authenticate(req, res, next) {
  const header = req.get('Authorization') ?? '';
  const [scheme, token] = header.split(' ');                 // "Bearer eyJ..." → ['Bearer', 'eyJ...']
  if (scheme !== 'Bearer' || !token) {
    res.set('WWW-Authenticate', 'Bearer');
    return res.status(401).json({ error: 'ต้องเข้าสู่ระบบก่อน' });
  }
  try {
    req.user = verifyToken(token);   // ผ่าน → แนบข้อมูลผู้ใช้ไว้ให้ middleware/controller ถัดไป
    next();
  } catch {
    res.set('WWW-Authenticate', 'Bearer error="invalid_token"');
    return res.status(401).json({ error: 'token ไม่ถูกต้องหรือหมดอายุ กรุณาเข้าสู่ระบบใหม่' });
  }
}

export function requireRole(role) {
  return (req, res, next) => {                        // คืน middleware — ใช้ซ้ำกับ role อื่นได้
    if (req.user?.role !== role) {
      return res.status(403).json({ error: 'ไม่มีสิทธิ์ทำรายการนี้' });
    }
    next();
  };
}
// api/src/routes/requestRoutes.js
import { authenticate, requireRole } from '../middleware/auth.js';

router.get('/', controller.listRequests);                          // ทุกคน
router.post('/', validateRequest, controller.createRequest);        // ทุกคน
router.get('/:id', controller.getRequest);                         // ทุกคน
router.put('/:id', authenticate, requireRole('staff'), controller.updateRequestStatus);
router.delete('/:id', authenticate, requireRole('staff'), controller.deleteRequest);

// frontend/src/services/authStore.js (เฉลย · ตัดมาบางส่วน)
const KEY = 'campus.auth.token';

export function getToken() {

    const token = localStorage.getItem(KEY);
