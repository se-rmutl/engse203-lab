ทางแก้ — cleanup function

4 นาที
useEffect(() => {
  let ignore = false;
  setLoadState('loading');

  getRequestById(requestId)
    .then((result) => {
      if (ignore) return;          ← ไม่เกี่ยวแล้ว ทิ้งไปเลย
      setRequest(result);
      setLoadState('success');
    })
    .catch((error) => {
      if (ignore) return;
      setErrorMessage(…);
      setLoadState('error');
    });

  return () => { ignore = true; };   ← React เรียกตอนออกจากหน้า
}, [requestId, reloadKey]);
จุดที่เข้าใจผิดง่ายที่สุด เราไม่ได้ยกเลิกคำขอ — คำขอยังวิ่งไปจนเสร็จตามปกติ ข้อมูลก็ยังถูกโหลดมาจริง
เราแค่เลือกไม่ใช้ผลลัพธ์ เท่านั้น
อธิบายเรื่อง cleanup function ถ้า Effect คืนฟังก์ชันกลับไป React จะเรียกฟังก์ชันนั้น ก่อนทำ Effect รอบใหม่ และ ก่อนที่ component จะถูกถอดออก

เฉลยคำถามที่ค้างมา

3 นาที
พูดว่า "ทีนี้กลับมาที่คำถามค้างจากสัปดาห์ที่แล้ว — React เรียก Effect 2 รอบตอนพัฒนา เพื่อทดสอบว่า Effect ของคุณรับมือกับการถูกเรียกซ้ำได้หรือไม่" 
เรียก Effect  →  เรียก cleanup  →  เรียก Effect อีกรอบ
ถ้าเขียนถูก เรียกซ้ำแล้วผลลัพธ์เหมือนเดิม ไม่มีอะไรผิดปกติ

ถ้าเขียนผิด

ข้อมูลซ้ำ ตัวจับเวลาซ้อนกัน การเชื่อมต่อค้าง — เห็นทันทีตอนพัฒนา ไม่ต้องรอผู้ใช้เจอ

ประโยคปิดที่อยากให้จำ "React ไม่ได้แค่บอกว่าคุณควรเขียน cleanup แต่ สร้างสถานการณ์ให้คุณเห็นปัญหาเองตั้งแต่ตอนพัฒนา — นี่คือแนวคิดการออกแบบเครื่องมือที่ดี ทำให้ความผิดพลาดปรากฏตัวเร็วที่สุด"
อย่าลืมย้ำ ตอน npm run build ขึ้นใช้งานจริง Effect จะทำงานรอบเดียว · พฤติกรรม 2 รอบมีเฉพาะตอนพัฒนาเท่านั้น
