async function fetchSeedRequests() {
  const baseUrl = import.meta.env?.BASE_URL ?? '/';
  const response = await fetch(`${baseUrl}data/initialRequests.json`);
  if (!response.ok) throw new Error('ไม่สามารถโหลดข้อมูลตัวอย่างได้');
  return structuredClone(await response.json());
}
สองจุดที่พลาดง่ายที่สุด หนึ่ง — ต้องมีเครื่องหมายคำถามใน import.meta.env?. ถ้าไม่ใส่ checker จะเรียกฟังก์ชันนี้นอก Vite ไม่ได้ แล้วรายงานว่าไม่ผ่าน ทั้งที่โค้ดทำงานปกติในเบราว์เซอร์

สอง — fetch() ไม่โยน error เมื่อเซิร์ฟเวอร์ตอบ 404 มันถือว่าติดต่อสำเร็จ เพียงแต่ได้หน้า error กลับมา ถ้าไม่ตรวจ response.ok แล้วเรียก .json() ต่อ คุณจะได้ error แปลก ๆ ว่าแปลง HTML เป็น JSON ไม่ได้ ซึ่งไม่บอกเลยว่าปัญหาจริงคือหาไฟล์ไม่เจอ

แล้วทำ TODO 5A-2 ต่อ — ลบบรรทัด throw ออกแล้วใส่บรรทัดนี้แทน

  return fetchSeedRequests();
  // TODO 5B-3: เปลี่ยนบรรทัดข้างบนเป็น return loadNormalRequests(options.onRecovery);

Effect ที่โหลดข้อมูล

useEffect(() => {
  setLoadState('loading');
  setErrorMessage('');
  setNotice('');

  getRequests({ scenario })
    .then((data) => {
      setRequests(data);
      setLoadState('success');
    })
    .catch((error) => {
      setErrorMessage(error instanceof Error ? error.message : 'เกิดข้อผิดพลาดที่ไม่ทราบสาเหตุ');
      setLoadState('error');
    });
  // TODO 5B: เพิ่ม cleanup guard เพื่อกัน stale update

คำตอบของคำถามตอนต้น ถ้าใส่ requests ลงไปด้วยจะเกิดวงวนไม่รู้จบ — Effect เซ็ต requests → requests เปลี่ยน → React เรียก Effect อีกรอบ → เซ็ตอีก วนไปเรื่อย ๆ

ทดสอบทั้ง 3 สถานะด้วย URL

URLต้องเห็น

ปุ่มลบใน CP นี้ยังทำงานกับหน่วยความจำอย่างเดียว

// คาบ 5A ลบได้เฉพาะในหน่วยความจำ — refresh แล้วกลับมาเหมือนเดิม
// TODO 5B-CP04a: เปลี่ยนเป็น await deleteRequest(requestId) เพื่อให้ลบแล้วหายจริง
function handleDelete(requestId) {
  setRequests((current) => current.filter((request) => request.id !== requestId));
  setNotice(`ลบคำร้อง ${requestId} ในหน่วยความจำแล้ว — refresh จะกลับมา`);
}
ทำไมยังไม่ลบจริง เพราะ deleteRequest() เป็นงานของคาบ 5B แต่ความสามารถ "ลบได้" เป็นของ Week 04 ที่ห้ามหาย
เราจึงให้ทำงานในหน่วยความจำไปก่อน พร้อมบอกผู้ใช้ตรง ๆ ว่า refresh แล้วจะกลับมา
