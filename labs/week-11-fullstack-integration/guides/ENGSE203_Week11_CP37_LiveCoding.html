health check ต้องเช็คถึงฐานข้อมูล ไม่ใช่แค่ว่า API เปิด

export function getDbStatus() {
  try {
    if (!db) return { connected: false, reason: 'ยังไม่ได้เปิดฐานข้อมูล' };
    const n = db.prepare("SELECT COUNT(*) c FROM sqlite_master WHERE type='table'").get().c;
    return { connected: true, driver: 'sqlite', tables: n };
  } catch (e) {
    return { connected: false, reason: e.message };
  }
}

healthRoutes.js

src/routes/

ตอบ 200 เมื่อปกติ · 503 เมื่อต่อ DB ไม่ได้

import { Router } from 'express';
import { getDbStatus } from '../services/requestService.js';
import { config } from '../config.js';
const router = Router();

router.get('/', (req, res) => {
  const db = getDbStatus();
  const ok = db.connected;
  res.status(ok ? 200 : 503).json({
    status: ok ? 'ok' : 'degraded',
    env: config.env, uptime: Math.round(process.uptime()),
    database: db, time: new Date().toISOString(),
  });
});
export default router;
