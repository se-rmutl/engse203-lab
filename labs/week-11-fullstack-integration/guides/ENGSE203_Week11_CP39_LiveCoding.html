app.js — API ใต้ /api · production เสิร์ฟหน้าเว็บ

api/src/app.js

Express ตรวจ route ตามลำดับ ตัวแรกที่ตรงชนะ — route / เดิมจึงชิงหน้าแรก

import path from 'node:path';
import { existsSync } from 'node:fs';

app.get('/api', (req, res) => res.json({ message: 'Campus Service API is running' }));
app.use('/api/health', healthRoutes);
app.use('/api/requests', requestRoutes);
app.use('/api/users', userRoutes);

if (config.isProd && existsSync(config.staticDir)) {
  app.use(express.static(config.staticDir));
  app.get(/^\/(?!api).*/, (req, res) =>
    res.sendFile(path.join(config.staticDir, 'index.html')));
} else {
  app.get('/', (req, res) => res.json({ message: 'API (dev) — หน้าเว็บอยู่ที่พอร์ต 5173' }));
}
(?!api) — ทุก path ที่ไม่ขึ้นต้นด้วย /api ได้ index.html ให้ React Router จัดการ · /api/nope ยังได้ JSON 404 เหมือนเดิม
