node:sqlite — เปิดฐานข้อมูลจากโค้ด

โมดูลที่มากับ Node 22 อยู่แล้ว ไม่ต้อง npm install อะไรเลย

🏫 เรียนและลงมือในห้อง≈ 11 นาที
บทที่ 2 / 9 · node:sqlite เครื่องมือที่ต้องรู้

เปิดฐานข้อมูลและสั่งงาน

import { DatabaseSync } from 'node:sqlite';

const db = new DatabaseSync('./data/campus.db');
db.exec('PRAGMA foreign_keys = ON');   // ⚠ ต้องสั่งทุกครั้ง
คำสั่งใช้เมื่อคืนอะไร
db.exec(sql)ไม่ต้องการผลลัพธ์—
.prepare(sql).all()หลายแถวarray
.prepare(sql).get()แถวเดียวobject หรือ undefined
.prepare(sql).run()INSERT · UPDATE · DELETE{ changes, lastInsertRowid }

result.changes บอกว่ากระทบกี่แถว — ถ้าเป็น 0 แปลว่าไม่พบ → คืน null → controller ตอบ 404
ไม่ต้อง SELECT เช็คก่อนแล้วค่อย UPDATE — สั่งครั้งเดียวแล้วดู changes พอ

บทที่ 2 / 9 · node:sqlite เหตุผลที่สำคัญ

ทำไมชื่อ DatabaseSync

Week 07 เรียก API ต้อง await ทุกครั้ง — ทำไมเรียกฐานข้อมูลไม่ต้อง

ผลดีที่ตามมา — controller ไม่ต้องแก้
เพราะ service ยังเป็น sync เหมือนเดิม · ถ้าเปลี่ยนเป็น async ทุกฟังก์ชัน controller จะต้องเติม await ทุกบรรทัด
นี่คือเหตุผลที่สัปดาห์นี้แก้ไฟล์เดียวได้จริง

สัปดาห์หน้า → MongoDB อยู่คนละเครื่องจริง ๆ ต้องรอข้อมูลข้ามเครือข่าย — ตรงนั้นจะได้เห็นว่าทำไมต้อง async

จะเห็น ExperimentalWarning — node:sqlite ยังเป็นฟีเจอร์ทดลองใน Node 22 · ไม่ใช่ error ซ่อนได้ด้วย --disable-warning=ExperimentalWarning ซึ่งใส่ใน package.json ให้แล้ว

จบบทที่ 2 — node:sqlite

อ่านข้อมูล — เอา JOIN จากสัปดาห์ที่แล้วมาใช้

ตรงนี้คือจุดที่ query ที่เขียนสัปดาห์ที่แล้วกลายเป็นโค้ดจริง

🏫 เรียนและลงมือในห้อง≈ 12 นาที
บทที่ 4 / 9 · อ่านข้อมูล สองรูปแบบไม่ตรงกัน

ฐานข้อมูลเก็บ id แต่ frontend ต้องการชื่อ

ฐานข้อมูลเก็บfrontend ต้องการ
ผู้แจ้งrequester_id: 1requesterName: "สมชาย ใจดี"
ประเภทrequest_typerequestType

และเราจะไม่แก้ frontend เพราะสัญญาไว้แล้ว

ทางออก — ใช้ JOIN + AS ที่เรียนไปแล้ว

const SELECT_SHAPE = `
  SELECT r.id,
         u.name          AS requesterName,
         r.request_type  AS requestType,
         r.location, r.details, r.priority, r.status
  FROM requests r
  JOIN users u ON u.id = r.requester_id`;

เก็บไว้เป็นค่าคงที่ตัวเดียว แล้วเอาไปต่อ WHERE ตามต้องการ — ไม่ต้องเขียน JOIN ซ้ำในทุกฟังก์ชัน

บทที่ 4 / 9 · อ่านข้อมูล โค้ดจริง

สองฟังก์ชันอ่านข้อมูล

export function findAll({ status } = {}) {
  return status
    ? db.prepare(`${SELECT_SHAPE} WHERE r.status = ? ORDER BY r.id`).all(status)
    : db.prepare(`${SELECT_SHAPE} ORDER BY r.id`).all();
}

export function findById(id) {
  return db.prepare(`${SELECT_SHAPE} WHERE r.id = ?`).get(id) ?? null;
}

สังเกตให้ดี — ${SELECT_SHAPE} ต่อได้ แต่ status ต่อไม่ได้
SELECT_SHAPE เป็นโค้ดของเราเองที่เขียนไว้ตายตัว · แต่ status มาจากผู้ใช้ จึงต้องส่งผ่าน ?
บทที่ 7 จะอธิบายว่าทำไม

⚠ ต้องมี ?? null — .get() คืน undefined เมื่อไม่พบ แต่ Week 07 สัญญาไว้ว่าจะคืน null · ถ้าลืม controller ยังตอบ 404 ได้ แต่ผิดสัญญาใน API contract

บทที่ 4 / 9 · อ่านข้อมูล

จบบทที่ 4 — อ่านข้อมูล

✓ สิ่งที่ต้องจำจากบทนี้

  • ค่าจากผู้ใช้ส่งผ่าน ? เสมอ ไม่ต่อเข้า SQL
  • .get() ต้องมี ?? null เพื่อรักษาสัญญาจาก Week 07

function resolveUserId(name) {
  const found = db.prepare('SELECT id FROM users WHERE name = ?').get(name);
  if (found) return found.id;          // มีแล้ว — ใช้ id เดิม

  const slug = Date.now().toString(36);
  return db.prepare('INSERT INTO users (name, department, email) VALUES (?,?,?)')
           .run(name, 'ไม่ระบุ', `user-${slug}@rmutl.ac.th`).lastInsertRowid;
}

lastInsertRowid คือ id ที่ฐานข้อมูลเพิ่งสร้างให้จาก AUTOINCREMENT
ทำไม email ต้องมี slug — เพราะ users.email มี UNIQUE ถ้าใส่ค่าเดิมทุกครั้งจะชนกันตั้งแต่คนที่สอง

สังเกตความสมมาตร
ขาเข้า — resolveUserId() แปลง ชื่อ → id ก่อนเก็บ
ขาออก — JOIN แปลง id → ชื่อ ก่อนส่งกลับ
service ทำหน้าที่เป็นล่ามระหว่างสองภาษา

CP29 · create แปลงชื่อเป็น id

export function create(input) {
  const id = nextId();
  db.prepare(`INSERT INTO requests
       (id, requester_id, request_type, location, details, priority)
     VALUES (?, ?, ?, ?, ?, ?)`)
    .run(id, resolveUserId(input.requesterName.trim()),
         input.requestType, input.location.trim(),
         input.details.trim(), input.priority ?? 'normal');
  return findById(id);   // คืนรูปแบบที่ frontend ต้องการ
}

✓ ผ่านเมื่อ — ทดสอบ 2 กรณี
① POST ด้วยชื่อที่มีอยู่แล้ว → ได้ 201 และ ตาราง users ไม่เพิ่ม
② POST ด้วยชื่อใหม่ → ได้ 201 และ ตาราง users เพิ่มคนใหม่

create() จบด้วย findById(id) เพื่อคืนรูปแบบที่มี requesterName ไม่ใช่ requester_id

07

Parameterized Query และ SQL Injection

ถึงตอนนี้เราใช้ ? มาตลอดโดยยังไม่ได้อธิบายว่าทำไม — บทนี้คือคำตอบ

🏠 เรียนในห้อง · ทำที่บ้าน≈ 12 นาที
บทที่ 7 / 9 · SQL Injection สาธิตให้เห็นจริง

ช่องโหว่ที่พบบ่อยที่สุดของเว็บ

// ❌ อันตราย
db.prepare(`SELECT * FROM requests WHERE status = '${status}'`).all();

ถ้าผู้ใช้ส่ง x' OR '1'='1 มา SQL จะกลายเป็น

SELECT * FROM requests WHERE status = 'x' OR '1'='1'

'1'='1' เป็นจริงเสมอ → คืนข้อมูลทั้งตารางออกไป แม้ผู้ใช้ไม่มีสิทธิ์เห็น
และถ้าส่ง '; DROP TABLE requests; -- มา — ตารางอาจหายทั้งตาราง

ต่อ string = ให้คนอื่นเติมข้อความลงกลางจดหมาย — เขาแทรกประโยคใหม่เข้ามาได้
parameterized = มีแบบฟอร์มที่เว้นช่องไว้ — กรอกได้แค่ในช่อง เขียนล้นออกมาเป็นคำสั่งไม่ได้

บทที่ 7 / 9 · SQL Injection กลไกการทำงาน

ฐานข้อมูลแยกคำสั่งกับข้อมูล

// ✅ ปลอดภัย
db.prepare('SELECT * FROM requests WHERE status = ?').all(status);
ต่อ stringparameterized
ฐานข้อมูลได้รับSQL ก้อนเดียวที่ปนค่าผู้ใช้โครง SQL + ค่า แยกกัน
ค่า x' OR '1'='1ตีความเป็นคำสั่งตีความเป็นข้อความธรรมดา
ผลลัพธ์คืนทั้งตารางคืน 0 แถว

? ใช้แทนได้เฉพาะ "ค่า" ไม่ใช่ชื่อคอลัมน์หรือตาราง
ORDER BY ? ใช้ไม่ได้ · ถ้าต้องให้ผู้ใช้เลือกคอลัมน์เรียง ต้องตรวจกับ allowlist ก่อน

🏠 CP31 — ยิง injection 3 แบบใส่ API ของตัวเอง แล้วพิสูจน์ว่าได้ 0 รายการทุกครั้ง และตารางยังอยู่

บทที่ 7 / 9 · SQL Injection ลองเล่น — ส่งค่าอันตรายเข้าไปดู

ต่อ string กับ parameterized ต่างกันแค่ไหน

✕ ต่อ string

— กดปุ่มด้านบน —

✓ parameterized (?)

— กดปุ่มด้านบน —

สังเกตฝั่งซ้าย — ค่าที่ผู้ใช้ส่งมากลายเป็นส่วนหนึ่งของคำสั่ง · ฝั่งขวาค่าถูกส่งแยกต่างหาก ฐานข้อมูลจึงมองเป็นข้อความธรรมดา

บทที่ 7 / 9 · SQL Injection

จบบทที่ 7 — SQL Injection

✓ สิ่งที่ต้องจำจากบทนี้

  • ต่อ string เข้า SQL = เปิดช่องให้แทรกคำสั่ง
  • ? ทำให้ฐานข้อมูลแยกคำสั่งกับข้อมูลออกจากกัน
  • ค่าที่แทรกมาถูกมองเป็นข้อความธรรมดา
  • ? ใช้แทนได้เฉพาะค่า · ถ้าต้องต่อ string ให้ใช้ allowlist

ฐานข้อมูลปฏิเสธข้อมูลผิดได้ — คำถามคือเมื่อถูกปฏิเสธแล้ว ผู้ใช้ควรเห็นอะไร

บทที่ 8 / 9 · Error จาก DB เหตุผลก่อน

constraint พัง แต่ใครผิด

errorใครผิดควรตอบ
FOREIGN KEY constraint failedผู้ใช้400
CHECK constraint failedผู้ใช้400
UNIQUE constraint failedผู้ใช้409
NOT NULL constraint failedผู้ใช้400

ถ้าไม่จัดการ errorHandler จะตอบ 500 ทั้งหมด — มีอะไรไม่ดี

500 แปลว่า "เซิร์ฟเวอร์ผิด" แต่กรณีเหล่านี้ส่วนใหญ่ผู้ใช้ส่งข้อมูลผิดมาเอง
① ผู้ใช้ไม่รู้ว่าต้องแก้อะไร เพราะ 500 บอกแค่ "ระบบพัง"
② ทีมจะเห็น 500 เยอะในระบบเฝ้าระวังแล้วตกใจ ทั้งที่ระบบปกติดี

ใช้ AppError ที่ทำไว้ตั้งแต่ Week 07 — ตอนนั้นเป็น Challenge ที่หลายคนข้าม · ตอนนี้ได้ใช้จริงแล้ว และ errorHandler ไม่ต้องแก้เลยเพราะมันอ่าน err.status อยู่แล้ว
